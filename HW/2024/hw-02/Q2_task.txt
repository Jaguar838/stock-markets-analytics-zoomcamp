def extract_ipo_data(url):
   response = requests.get(url, headers=headers)
   # Wrap the response text in a StringIO object
   content = StringIO(response.text)
   # Read the HTML tables using read_html
   ipo_dfs = pd.read_html(content)
   return ipo_dfs[0]

# convert to datetime
def convert_columns(df):
   # Конвертує 'IPO Date'
   formats = ['mixed', '%m/%d/%Y', '%d-%b-%Y', None]
   for fmt in formats:
       try:
           df['IPO Date'] = pd.to_datetime(df['IPO Date'], format=fmt)
           break
       except ValueError:
           continue
   # Усуває дефіси та перетворює на верхній регістр
   df['Symbol'] = df['Symbol'].str.replace('-', '').str.upper()
   return df

def process_symbol_column(df):
   # Зіставлення символів
   symbol_mapping = {

       'IBAC': 'IBACU', # Приклад

       # Додайте інші відповідності тут

   }
   # Список небажаних тікетів
   symbols_to_drop = ['RYZB', 'PTHR'] #, 'JVSA', , 'BUYA', 'BAYA', 'FORL', 'YGFGF', 'LEGT', 'DYCQ']
   # Видалення рядків, де 'Symbol' є у списку тікетів для вилучення
   df = df.drop(df[df['Symbol'].isin(symbols_to_drop)].index)
   # Заміна символів за зіставленням
   df['Symbol'] = df['Symbol'].replace(symbol_mapping)
   return df.loc[df['IPO Date'] <= '2024-03-01']

def get_daily_prices(ipo_df):
   # Отримати масив символів з датафрейму IPO
   symbols = ipo_df['Symbol'].tolist()[::-1]
   # Отримати всі денні ціни 'Adj Close' для вказаних символів
   stocks_df = yf.download(" ".join(symbols), period='max', interval="1d")['Adj Close']
   start=ipo_df['IPO Date'].min()
   end=ipo_df['IPO Date'].max()
   return start, end, symbols, stocks_df.query('index >= @start')

### 3) Уточніть дати IPO по данним сайту Yahoo Finance та створіть таблицю min_dates (перший день, коли кожна акція має дані на Yahoo Finance)

def get_min_dates(daily_prices_df):
   min_dates = []
   for ticker in daily_prices_df.columns:
       # Шукаємо перший непорожній рядок у стовпці
       ipo_date = daily_prices_df[ticker].first_valid_index()
       min_dates.append({'ticker': ticker, 'ipo_date': ipo_date})
   return pd.DataFrame(min_dates)
